# DetailLayout

The `DetailLayout` component provides a full-page layout with a fixed breadcrumb header, a left sidebar with menu items (supporting nested items and dividers), and a main content area. Use it for detail pages, resource dashboards, or any screen that needs structured navigation and breadcrumbs.

## Usage

```tsx
import { DetailLayout } from 'tessera-ui/components/layouts'
import type { DetailItemsProps, BreadcrumbItemData } from 'tessera-ui/components/layouts/types'
import { Home, FileText, Settings } from 'lucide-react'

const menuItems: DetailItemsProps[] = [
  { title: 'Overview', path: '/overview', icon: Home, disabled: true },
  { title: 'Documents', path: '/documents', icon: FileText },
  { title: 'Settings', path: '/settings', icon: Settings, divider: true },
]

const breadcrumbs: BreadcrumbItemData[] = [
  { label: 'Home', link: '/' },
  { label: 'Project', link: '/project' },
  { label: 'Overview', link: '/project/overview', disabled: true },
]

function ProjectOverviewPage() {
  return (
    <DetailLayout menuItems={menuItems} breadcrumbs={breadcrumbs} isLoading={false}>
      <div className="p-4">
        <h1>Overview</h1>
        <p>Your main content here.</p>
      </div>
    </DetailLayout>
  )
}
```
