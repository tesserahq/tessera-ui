# Header

The `Header` component is the main application header bar. It includes the app switcher (AppMenu), logo/title link, optional left/center/right content slots, and the profile menu with theme switching and logout.

## Usage

```tsx
import { Header } from 'tessera-ui/components/layouts/main/header'
import { TesseraProvider } from 'tessera-ui/main'
import type { AppHostUrls } from 'tessera-ui/main'

function AppShell() {
  return (
    <TesseraProvider identiesApiUrl="" token="">
      <Header
        title="Custos"
        selectedTheme="system"
        onSetTheme={(theme) => {}}
        actionLogout={() => {}}
        actionProfile={() => {}}
        isStorybook={false}
      />
      <main className="pt-[60px]">Your page content</main>
    </TesseraProvider>
  )
}
```
