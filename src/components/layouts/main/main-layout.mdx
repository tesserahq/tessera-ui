# MainLayout

The `MainLayout` component provides the primary application shell: a collapsible sidebar navigation and a main content area. It wraps content in `SidebarProvider` and renders `SidebarPanel` plus a `SidebarInset` for the page body.

## Usage

```tsx
import { MainLayout } from 'tessera-ui/components/layouts/main'
import { Header } from 'tessera-ui/components/layouts/main/header'
import type { MainItemProps } from 'tessera-ui/components/layouts/types'
import { KeyRound, Settings, Users } from 'lucide-react'

const menuItems: MainItemProps[] = [
  { title: 'Roles', path: '/roles', icon: KeyRound },
  { title: 'Users', path: '/users', icon: Users },
  { title: 'Settings', path: '/settings', icon: Settings },
]

function App() {
  const collapseSidebar = false // e.g. derive from route depth

  return (
    <>
      <Header title="Custos" {...headerProps} />
      <MainLayout menuItems={menuItems} collapseSidebar={collapseSidebar}>
        <div className="p-6">Your page content or nested DetailLayout</div>
      </MainLayout>
    </>
  )
}
```
